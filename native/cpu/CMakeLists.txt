cmake_minimum_required(VERSION 3.22)
project(aikernel_cpu_accel C)
# A single-config build without a type must not silently benchmark unoptimized code.
# Explicit Debug and multi-config generators remain available for diagnostics.
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64")
  message(FATAL_ERROR "CPU accelerator currently targets x64 only")
endif()
add_library(aikernel_cpu_accel SHARED dispatch.c vnni.c avxvnni.c avx512vnni.c avx2int8.c amx.c)
set_target_properties(aikernel_cpu_accel PROPERTIES C_STANDARD_REQUIRED YES
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(aikernel_cpu_accel PRIVATE /W4 /WX /fp:precise)
  set_source_files_properties(avxvnni.c PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
  set_source_files_properties(avx2int8.c PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
  set_source_files_properties(avx512vnni.c PROPERTIES COMPILE_OPTIONS "/arch:AVX512")
else()
  target_compile_options(aikernel_cpu_accel PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  target_link_libraries(aikernel_cpu_accel PRIVATE m)
endif()
