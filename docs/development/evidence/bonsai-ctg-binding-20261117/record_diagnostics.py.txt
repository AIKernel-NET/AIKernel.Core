from pathlib import Path
import json,hashlib
root=Path(__file__).parent
repo=Path(r'D:/AIKernel/artifacts/cuda-driver-native/core-context-free-pci')
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
paths=['tests/AIKernel.Inference.Optimization.Validation/Program.cs','tests/AIKernel.Inference.Optimization.Validation/BonsaiCtgReadingProbe.cs']
data={'sourceSha256':{p:sha(repo/p) for p in paths},'assemblySha256':{p.name:sha(p) for p in (root/'app-diagnostics').glob('AIKernel*.dll')}}
original=json.loads((root/'tested-build.json').read_text(encoding='utf-8'))
assert all(digest==data['assemblySha256'][name] for name,digest in original['assemblySha256'].items() if name!='AIKernel.Core.Tests.dll')
data['kernelAssembliesMatchTestedBuild']=True
(root/'diagnostics-build.json').write_text(json.dumps(data,indent=2)+'\n',encoding='utf-8')
snapshot=root/'diagnostics-sources';snapshot.mkdir(exist_ok=True)
for i,p in enumerate(paths):(snapshot/(str(i)+'.cs.txt')).write_bytes((repo/p).read_bytes())
(snapshot/'paths.json').write_text(json.dumps(paths,indent=2)+'\n',encoding='utf-8')
print('Diagnostics build recorded; all kernel assemblies match the tested build.')
