from pathlib import Path
import json,hashlib,subprocess,re
repo=Path(r'D:/AIKernel/artifacts/cuda-driver-native/core-context-free-pci');folder=repo/'docs/development/evidence/bonsai-ctg-reading-20261117'
sha=lambda b:hashlib.sha256(b).hexdigest();read=lambda p:json.loads((folder/p).read_text(encoding='utf-8'))
manifest=read('evidence-sha256.json')
for name,digest in manifest.items():
 p=folder/name;assert sha(p.read_bytes())==digest,name
 assert sha(subprocess.check_output(['git','show',':'+p.relative_to(repo).as_posix()],cwd=repo))==digest,('staged',name)
build=read('focused-tested-build.json')
for name,digest in build['sourceSha256'].items():
 content=(repo/name).read_bytes();assert sha(content)==digest,name
 staged=subprocess.check_output(['git','show',':'+name],cwd=repo)
 assert staged.replace(b'\r\n',b'\n')==content.replace(b'\r\n',b'\n'),('staged source',name)
for label in ('diagnostic','focused'):
 tested=read(label+'-tested-build.json');source=folder/('sources-'+label)
 for i,path in enumerate(json.loads((source/'paths.json').read_text(encoding='utf-8'))):
  assert sha((source/(str(i)+'.cs.txt')).read_bytes())==tested['sourceSha256'][path]
for run in folder.glob('server-*'):
 if not run.is_dir():continue
 transferred=json.loads((run/'transferred-files.json').read_text(encoding='utf-8'))
 label='diagnostic' if run.name in ('server-reading-matrix','server-grounding-matrix','server-grounding-reordered') else 'focused'
 tested=read(label+'-tested-build.json')
 for name,digest in tested['assemblySha256'].items():assert transferred['app/'+name]==digest,(run.name,'assembly',name)
 for name,digest in tested['canonicalRomSha256'].items():assert transferred['governance/'+name]==digest,(run.name,'ROM',name)
 for name,digest in transferred.items():
  p=run/(name+'.txt' if name.endswith('.py') else name)
  if p.exists():assert sha(p.read_bytes())==digest,(run.name,name)
report=repo/'docs/development/bonsai-ctg-reading-20261117-jp.md';text=report.read_text(encoding='utf-8');assert 'RESULTS_PENDING' not in text
for link in re.findall(r'\]\(([^)]+)\)',text):assert (report.parent/link).exists(),link
print(str(len(manifest))+' artifacts and staged files match; source snapshots, actual transfers and links verified.')
