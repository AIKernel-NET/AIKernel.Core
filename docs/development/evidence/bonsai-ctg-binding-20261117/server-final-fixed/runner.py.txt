from pathlib import Path
import os,subprocess,json,time,resource
out=Path(__file__).resolve().parent;root=out.parent.parent
os.umask(0o077)
dependency=out.parent/'host-guards';deadline=time.time()+7200
while not (dependency/'completion.json').exists():
 if time.time()>deadline:raise RuntimeError('Preceding run has not completed')
 time.sleep(5)
previous=json.loads((dependency/'completion.json').read_text())
assert previous['stopReason'] is None and previous['exitCode'] in (0,1)
rows=json.loads((dependency/'result.json').read_text())
assert len(rows)==7 and all(r['expectedHostBlock'] and r['pipelinePass'] and r['inferenceCalls']==0 for r in rows)
assert {4,5}.issubset(os.sched_getaffinity(0))
os.sched_setaffinity(0,{4,5});os.nice(10)
resource.setrlimit(resource.RLIMIT_AS,(8*1024**3,8*1024**3))
env=os.environ.copy();env.update(DOTNET_PROCESSOR_COUNT='2',DOTNET_GCHeapHardLimit='0x80000000')
runtime=Path(json.loads((root/'runtime.lock.json').read_text())['path'])/'dotnet'
model=root/'models/bonsai1.7b/Bonsai-1.7B-Q1_0.gguf'
commands={
 'evidence-probe':[str(runtime),str(out/'app/AIKernel.Core.Tests.dll'),'--bonsai-ctg-evidence-probe',str(model),str(out/'governance'),str(out/'cases.json'),str(out/'evidence-probe.json')],
 'normal-128':[str(runtime),str(out/'app/AIKernel.Core.Tests.dll'),str(model),'cpu','avx512']}
checks={}
for name,command in commands.items():
 peak=0;reason=None;started=time.time()
 with (out/(name+'.log')).open('wb') as log:
  child=subprocess.Popen(command,env=env,stdout=log,stderr=subprocess.STDOUT,stdin=subprocess.DEVNULL)
  (out/'pid.json').write_text(json.dumps({'pid':child.pid,'runnerPid':os.getpid(),'check':name,'cpus':[4,5]}))
  while child.poll() is None:
   try:
    resident=int((Path('/proc')/str(child.pid)/'statm').read_text().split()[1])*os.sysconf('SC_PAGE_SIZE');peak=max(peak,resident)
    if resident>7*1024**3:reason='rss-limit'
   except (OSError,ValueError,IndexError):pass
   if time.time()-started>1200:reason='check-deadline'
   if reason:
    child.terminate()
    try:child.wait(timeout=15)
    except subprocess.TimeoutExpired:child.kill();child.wait()
    break
   time.sleep(1)
  checks[name]={'exitCode':child.wait(),'maximumRssBytes':peak,'stopReason':reason,'seconds':time.time()-started}
 (out/'checks.json').write_text(json.dumps(checks,indent=2))
(out/'completion.json').write_text(json.dumps({'checks':checks,'cpuAffinity':[4,5],'configuredWorkers':4,'performanceBenchmark':False},indent=2))
