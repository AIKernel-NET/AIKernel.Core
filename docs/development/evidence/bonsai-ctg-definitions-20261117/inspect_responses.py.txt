from pathlib import Path
import sys,json,subprocess
sys.path.insert(0,r'D:/AIKernel/.wp-predelete-backup')
from xserver_ssh import connection
phase=sys.argv[1]
assert phase in ('scope-pilot', 'provenance-pilot', 'scope-full', 'provenance-full', 'host-guards', 'final-api', 'reading-probe', 'criteria-pilot', 'criteria-full', 'final-v18','criteria-pilot-fixed','focus-pilot','focus-full','effects-pilot')
remote='''from pathlib import Path
import json
out=Path.home()/'.aikernel-inquiry-014/ctg-definitions-20261117'/PHASE
result={}
for name in ('result.json','result.json.responses.json'):
 if (out/name).exists():
  try:result[name]=json.loads((out/name).read_text())
  except ValueError:pass
print(json.dumps(result))
'''.replace('PHASE',repr(phase))
with connection() as ssh:
 p=subprocess.run(ssh+['/usr/bin/python3.6 -'],input=remote.encode(),capture_output=True,timeout=90,creationflags=subprocess.CREATE_NO_WINDOW)
assert p.returncode==0,p.stderr.decode(errors='replace')[-1000:]
data=json.loads(p.stdout)
(Path(__file__).parent/(phase+'-snapshot.json')).write_text(json.dumps(data,indent=2)+'\n',encoding='utf-8')
for row in data.get('result.json.responses.json',[])[-6:]:print(json.dumps({'id':row['id'],'role':row['council'],'stage':row.get('stage'),'criterion':row.get('criterion'),'text':row['answer']['Text']},ensure_ascii=True))
for row in data.get('result.json',[]):print(json.dumps({'id':row['id'],'pass':row['pass'],'error':row['result']['Evaluation']['ErrorCode'],'checks':row['checks']},ensure_ascii=True))
