import json
from pathlib import Path

ROOT = Path(__file__).parent
REPO = Path(r'D:\AIKernel\artifacts\cuda-driver-native\core-context-free-pci')
OLD = REPO / 'docs/development/evidence/bonsai-ctg-refine-20261116'
FINDINGS = REPO / 'docs/development/evidence/bonsai-ctg-findings-20261116'
def read(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))
def write(name, value):
    (ROOT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')

fixtures = read(REPO / 'tests/AIKernel.Inference.Optimization.Validation/fixtures/bonsai-ctg-separated-20261116.json')
by_id = {row['id']: row for row in fixtures}
baseline = read(OLD / 'candidate-existing.json') + read(OLD / 'candidate-holdout.json') + read(FINDINGS / 'baseline-fresh.json')
if (ROOT / 'baseline-pairs.json').exists():
    baseline += read(ROOT / 'baseline-pairs.json')
candidate = read(ROOT / 'candidate.json') if (ROOT / 'candidate.json').exists() else []

def metrics(rows):
    return dict(count=len(rows), votesPass=sum(row['pass'] for row in rows),
                gatePass=sum(row['result']['Gate']['Accepted'] == row['expectedGate'] for row in rows),
                evaluationFailures=sum(not row['result']['Evaluation']['Succeeded'] for row in rows),
                falseAllow=sum(row['result']['Gate']['Accepted'] and not row['expectedGate'] for row in rows),
                falseDeny=sum(not row['result']['Gate']['Accepted'] and row['expectedGate'] for row in rows))
summary = {}
for label, rows in [('baseline', baseline), ('candidate', candidate)]:
    summary[label] = metrics(rows)
    summary[label + '-groups'] = {group: metrics([row for row in rows if by_id[row['id']]['group'] == group])
                                 for group in ('existing', 'previous-holdout', 'findings-holdout', 'new-pairs')}
failures = []
for row in candidate:
    if not row['pass']:
        failures.append(dict(id=row['id'], expectedGate=row['expectedGate'], gate=row['result']['Gate']['Accepted'],
                             checks=row['checks'], error=row['result']['Evaluation']['ErrorCode']))
old_by_id = {row['id']: row for row in baseline}
completed_ids = {row['id'] for row in candidate if row['result']['Evaluation']['Succeeded']
                 and old_by_id[row['id']]['result']['Evaluation']['Succeeded']}
matched_completed = {label: metrics([row for row in rows if row['id'] in completed_ids])
                     for label, rows in [('baseline', baseline), ('candidate', candidate)]}
changes = [dict(id=row['id'], before=old_by_id[row['id']]['pass'], after=row['pass']) for row in candidate
           if row['id'] in old_by_id and old_by_id[row['id']]['pass'] != row['pass']]
write('comparison.json', dict(summary=summary, matchedCompleted=matched_completed, failures=failures, changes=changes))

findings = sum([read(FINDINGS / (name + '.json')) for name in ('candidate-existing-v3c', 'candidate-holdout-v3c', 'candidate-fresh-v3c')], [])
citations = []
for row in findings:
    facts = dict(by_id[row['id']]['facts'], proposal=by_id[row['id']]['proposal'])
    keys = sorted(facts)
    metadata = row['result']['Evaluation']['Metadata']
    for role in ('Logos', 'Ethos', 'Pathos'):
        key = role + '.evidence.source'
        if key in metadata and keys[int(metadata[key])] == 'proposal':
            citations.append(dict(id=row['id'], council=role, finding=metadata[role + '.finding'],
                                  gate=row['result']['Gate']['Accepted'], expectedGate=row['expectedGate']))
write('prior-proposal-citations.json', dict(totalCouncilResponses=90, proposalCitations=citations,
      proposalCitationCount=len(citations), falseAllowCaseIds=sorted({row['id'] for row in citations if row['gate'] and not row['expectedGate']}),
      interpretation='A source reference to the candidate proposal was structurally accepted. This does not establish that it caused every semantic error.'))
print(json.dumps(dict(summary=summary, failures=failures), ensure_ascii=False, indent=2))
