from pathlib import Path
import hashlib,json,re,subprocess
repo=Path(r'D:/AIKernel/artifacts/cuda-driver-native/core-context-free-pci')
folder=repo/'docs/development/evidence/bonsai-ctg-overrefusal-20261117'
manifest=json.loads((folder/'evidence-sha256.json').read_text(encoding='utf-8'))
for name,digest in manifest.items():
 path=folder/name
 assert hashlib.sha256(path.read_bytes()).hexdigest()==digest,name
 staged=subprocess.check_output(['git','show',':'+path.relative_to(repo).as_posix()],cwd=repo)
 assert hashlib.sha256(staged).hexdigest()==digest,('staged',name)
build=json.loads((folder/'guarded-tested-build.json').read_text(encoding='utf-8'))
for name,digest in build['sourceSha256'].items():
 assert hashlib.sha256((repo/name).read_bytes()).hexdigest()==digest,name
for run in folder.glob('server-*'):
 if not run.is_dir():continue
 for name,digest in json.loads((run/'transferred-files.json').read_text()).items():
  path=run/(name+'.txt' if name.endswith('.py') else name)
  if path.exists():assert hashlib.sha256(path.read_bytes()).hexdigest()==digest,(run.name,name)
report=repo/'docs/development/bonsai-ctg-overrefusal-20261117-jp.md'
text=report.read_text(encoding='utf-8');assert 'RESULTS_PENDING' not in text
for link in re.findall(r'\]\(([^)]+)\)',text):assert (report.parent/link).exists(),link
print(str(len(manifest))+' artifacts match working-tree and staged hashes; final source, transferred evidence and report links match.')
